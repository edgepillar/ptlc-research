// Original offline PTLC quote probe over existing public synthetic originals.
// Source-test stores select synthetic feature flags, never node activation.
package embedded

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"math/big"
	"os"
	"strconv"
	"testing"

	"github.com/zenon-network/go-zenon/chain/nom"
	"github.com/zenon-network/go-zenon/common/types"
	"github.com/zenon-network/go-zenon/vm/constants"
)

type researchPtlcQuoteCache struct{ requiredPoWQuoteCache }

func (s *researchPtlcQuoteCache) IsSporkActive(spork *types.ImplementedSpork) (bool, error) {
	if spork == types.PtlcSpork {
		return true, nil
	}
	return s.requiredPoWQuoteCache.IsSporkActive(spork)
}

func TestResearchNoMPtlcRpcQuoteCorpus(t *testing.T) {
	input, err := os.ReadFile(os.Getenv("PTLC_NOM_RPC_PTLC_FIXTURE"))
	if err != nil {
		t.Fatal("selected public originals unavailable")
	}
	var corpus map[string]struct {
		Wire   string `json:"wire_hex"`
		SHA256 string `json:"wire_sha256"`
		Hash   string `json:"hash_hex"`
	}
	if json.Unmarshal(input, &corpus) != nil {
		t.Fatal("selected public record set refused")
	}
	for _, leg := range []string{"long", "short"} {
		original, ok := corpus[leg+"-original"]
		if !ok {
			t.Fatal("selected public leg unavailable")
		}
		raw, err := hex.DecodeString(original.Wire)
		if err != nil || len(raw) == 0 || len(raw) > 2048 {
			t.Fatal("bounded public original required")
		}
		digest := sha256.Sum256(raw)
		if hex.EncodeToString(digest[:]) != original.SHA256 {
			t.Fatal("public original digest differs")
		}
		for _, tc := range []struct {
			id                           string
			active                       bool
			version, fusion, work, units uint64
		}{
			{"dynamic-fused-only", true, 2, 1050, 1100, 40},
			{"dynamic-partial-ceil", true, 2, 1051, 1103, 30},
			{"dynamic-no-fusion", true, 2, 1051, 1103, 0},
			{"before-first-v2-covered", true, 1, 1051, 1103, 40},
			{"before-first-v2-partial", true, 1, 1051, 1103, 30},
			{"legacy-covered", false, 1, 0, 0, 40},
			{"legacy-partial", false, 1, 0, 0, 30},
			{"legacy-no-fusion", false, 1, 0, 0, 0},
		} {
			t.Run(leg+"-"+tc.id, func(t *testing.T) {
				block, err := nom.DeserializeAccountBlock(raw)
				if err != nil || block.BlockType != nom.BlockTypeUserSend || block.ToAddress != types.PtlcContract {
					t.Fatal("selected public PTLC send refused")
				}
				if hex.EncodeToString(block.Hash.Bytes()) != original.Hash || block.ComputeHash() != block.Hash {
					t.Fatal("selected node identifier differs")
				}
				frontier := &nom.Momentum{Version: tc.version, Height: 100,
					NextFusionPrice: tc.fusion, NextWorkPrice: tc.work}
				fused := new(big.Int).Mul(new(big.Int).SetUint64(tc.units), new(big.Int).SetUint64(constants.CostPerFusionUnit))
				api := &PlasmaApi{chain: &requiredPoWQuoteChain{
					momentum: &requiredPoWQuoteMomentum{frontier: frontier},
					account:  &requiredPoWQuoteAccount{address: block.Address},
					cache:    &researchPtlcQuoteCache{requiredPoWQuoteCache{active: tc.active, fused: fused}},
				}}
				result, err := api.GetRequiredPoWForAccountBlock(GetRequiredParam{
					SelfAddr: block.Address, ToAddr: &block.ToAddress, BlockType: block.BlockType, Data: block.Data,
				})
				if err != nil || result == nil {
					t.Fatal("selected unsigned PTLC quote refused")
				}
				encodedResult, err := json.Marshal(result)
				if err != nil {
					t.Fatal("selected typed PTLC result refused")
				}
				record := map[string]interface{}{
					"schema": "ptlc-nom-ptlc-rpc-quote-source-record-v1",
					"id":     leg + "-" + tc.id, "kind": "source-ptlc-unlock-rpc",
					"input": map[string]interface{}{
						"leg": leg, "original_wire_sha256": original.SHA256, "original_node_hash_hex": original.Hash,
						"caller_hex":   hex.EncodeToString(block.Address.Bytes()),
						"contract_hex": hex.EncodeToString(block.ToAddress.Bytes()),
						"block_type":   strconv.FormatUint(block.BlockType, 10), "data_hex": hex.EncodeToString(block.Data),
						"ptlc_spork_enforced": true, "dynamic_spork_enforced": tc.active,
						"frontier_version":  strconv.FormatUint(tc.version, 10),
						"next_fusion_price": strconv.FormatUint(tc.fusion, 10),
						"next_work_price":   strconv.FormatUint(tc.work, 10), "fusion_units": strconv.FormatUint(tc.units, 10),
					},
					"typed_result_json_hex": hex.EncodeToString(encodedResult),
				}
				encoded, err := json.Marshal(record)
				if err != nil {
					t.Fatal("selected PTLC projection refused")
				}
				t.Logf("PTLC_NOM_PTLC_RPC_QUOTE_RECORD=%s", encoded)
			})
		}
	}
}
